KEYBOARD + SCROLL NAVIGATION JS:
const fpContainer = document.getElementById('fullpage');
const sections = [...document.querySelectorAll('.fp-section')];
let current = 0;

// Build sidenav
sections.forEach((sec, i) => {
  const dot = document.createElement('div');
  dot.className = 'fp-nav-dot' + (i === 0 ? ' active' : '');
  dot.innerHTML = `
    <span class="fp-nav-dot-label">${sec.dataset.nav}</span>
    <div class="fp-nav-dot-circle"></div>`;
  dot.onclick = () => goToSection(i);
  document.getElementById('fpSidenav').appendChild(dot);
});

function goToSection(index) {
  sections[index].scrollIntoView({ behavior: 'smooth' });
}

function updateNav(index) {
  current = index;
  document.querySelectorAll('.fp-nav-dot')
    .forEach((d,i) => d.classList.toggle('active', i === index));
  const pad = n => String(n).padStart(2,'0');
  document.getElementById('fpCurrent').textContent = pad(index+1);
  const pct = (index / (sections.length-1)) * 100;
  document.getElementById('fpProgress').style.height = pct + '%';
}

// IntersectionObserver for active section detection
const observer = new IntersectionObserver(entries => {
  entries.forEach(e => {
    if (e.isIntersecting && e.intersectionRatio > 0.5) {
      const i = sections.indexOf(e.target);
      updateNav(i);
      animateSectionIn(e.target);
    }
  });
}, { threshold: 0.5 });
sections.forEach(s => observer.observe(s));

// Keyboard navigation
document.addEventListener('keydown', e => {
  if (e.key === 'ArrowDown' || e.key === 'PageDown') {
    e.preventDefault();
    goToSection(Math.min(current+1, sections.length-1));
  }
  if (e.key === 'ArrowUp' || e.key === 'PageUp') {
    e.preventDefault();
    goToSection(Math.max(current-1, 0));
  }
});

// Section entrance animations
function animateSectionIn(section) {
  section.querySelectorAll('[data-animate]').forEach((el, i) => {
    el.style.animationDelay = (i * 0.12) + 's';
    el.classList.add('animate-in');
  });
}

}

}

  to { opacity: 1; transform: translateY(0); }
}

PART 3: BACKGROUND SYSTEM — Video, Image, Gradient
═══════════════════════════════════════════════

Every section can have its own background type.
The CMS controls this with zero code.

BACKGROUND TYPES:
1. VIDEO (looping, muted, autoplay)
2. IMAGE (with parallax option)
3. GRADIENT (CSS gradient builder)
4. PARTICLE ANIMATION (pure CSS)
5. GALLERY SLIDESHOW (images rotate)
6. SOLID COLOR

BACKGROUND RENDERER:
function renderBackground(sectionId, bgConfig) {
  const section = document.getElementById(sectionId);
  const bgEl = section.querySelector('.fp-bg');
  if (!bgEl) return;
  
  bgEl.innerHTML = '';
  bgEl.className = 'fp-bg';
  
  switch(bgConfig.type) {
    
    case 'video':
      bgEl.innerHTML = `
        <video autoplay muted loop playsinline 
               style="width:100%;height:100%;object-fit:cover">
          <source src="${bgConfig.videoUrl}" type="video/mp4">
        </video>`;
      break;
      
    case 'image':
      bgEl.style.backgroundImage = `url('${bgConfig.imageUrl}')`;
      bgEl.style.backgroundSize = 'cover';
      bgEl.style.backgroundPosition = 'center';
      if (bgConfig.parallax) {
        bgEl.style.backgroundAttachment = 'fixed';
      }
      break;
      
    case 'gallery':
      // Auto-rotating gallery as background
      let idx = 0;
      const imgs = bgConfig.images || [];
      bgEl.style.transition = 'opacity 1.5s ease';
      function rotateBg() {
        bgEl.style.opacity = 0;
        setTimeout(() => {
          bgEl.style.backgroundImage = `url('${imgs[idx % imgs.length]}')`;
          bgEl.style.backgroundSize = 'cover';
          bgEl.style.backgroundPosition = 'center';
          bgEl.style.opacity = 1;
          idx++;
        }, 1500);
      }
      bgEl.style.backgroundImage = `url('${imgs[0]}')`;
      bgEl.style.backgroundSize = 'cover';
      bgEl.style.backgroundPosition = 'center';
      setInterval(rotateBg, bgConfig.interval || 5000);
      break;
      
    case 'gradient':
      bgEl.style.background = bgConfig.gradient;
      // e.g. "linear-gradient(135deg, #080810 0%, #1a0a2e 100%)"
      break;
      
    case 'particles':
      // Pure CSS animated particles — no library
      renderCSSParticles(bgEl, bgConfig);
      break;
  }
  
  // Apply overlay opacity from CMS
  const overlay = section.querySelector('.fp-bg-overlay');
  if (overlay && bgConfig.overlayOpacity !== undefined) {
    overlay.style.background = 
      `rgba(8,8,16,${bgConfig.overlayOpacity})`;
  }
}

// CSS Particles (pure CSS, no JS library):
function renderCSSParticles(container, config) {
  const count = config.count || 30;
  const color = config.color || 'rgba(200,168,75,0.3)';
  let html = '<style>';
  for (let i = 0; i < count; i++) {
    const x = Math.random() * 100;
    const delay = Math.random() * 8;
    const duration = 6 + Math.random() * 8;
    const size = 2 + Math.random() * 4;
    html += `
      .particle-${i} {
        position:absolute;
        left:${x}%;bottom:-10px;
        width:${size}px;height:${size}px;
        background:${color};border-radius:50%;
        animation:rise ${duration}s ${delay}s infinite linear;
      }`;
  }
  html += `
    @keyframes rise {
      0%{transform:translateY(0) scale(1);opacity:0}
      10%{opacity:1}
      90%{opacity:0.5}
      100%{transform:translateY(-110vh) scale(0.3);opacity:0}
    }
  </style>`;
  for (let i = 0; i < count; i++) {
    html += `<div class="particle-${i}"></div>`;
  }
  container.innerHTML = html;
}
