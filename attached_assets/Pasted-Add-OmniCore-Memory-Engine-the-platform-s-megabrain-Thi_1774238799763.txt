CREATE TABLE omnicore_memories (
  id TEXT PRIMARY KEY,
  -- WHAT TYPE OF MEMORY
  memory_type TEXT NOT NULL,
  -- Types:
  -- niche_keyword: a keyword that converts in a specific niche
  -- pricing_pattern: a price point that works for a product type
  -- image_pattern: an image type/style that converts
  -- prompt_template: a Claude prompt that produces great results
  -- competitor_intel: competitor data for a niche
  -- ab_insight: what wins in A/B tests per niche
  -- seasonal_pattern: seasonal sales patterns per niche
  -- product_gap: product types missing in a niche
  -- supplier_intel: supplier info for a product type
  
  -- CONTEXT
  niche TEXT,           -- moda, tecnologia, deporte, hogar, belleza...
  sub_niche TEXT,       -- moda_urbana, running, gaming...
  product_type TEXT,    -- chaqueta, zapatillas, auriculares...
  market TEXT,          -- es, mx, co, latam, global
  
  -- THE MEMORY CONTENT
  title TEXT NOT NULL,
  content TEXT NOT NULL,      -- the actual knowledge (JSON or text)
  confidence REAL DEFAULT 0.5, -- 0.0-1.0, increases with validation
  use_count INTEGER DEFAULT 0, -- how many times applied
  success_count INTEGER DEFAULT 0, -- how many times it worked
  success_rate REAL DEFAULT 0.0,
  
  -- SOURCE (anonymized, never reveals which client)
  source_type TEXT,  -- manual | learned_audit | learned_ab | 
                     -- learned_pricing | learned_image | scraped
  
  -- METADATA
  tags TEXT,         -- JSON array of tags for fast filtering
  embedding TEXT,    -- future: vector embedding for semantic search
  created_at TEXT DEFAULT (datetime('now')),
  last_used_at TEXT,
  last_updated_at TEXT DEFAULT (datetime('now')),
  is_verified INTEGER DEFAULT 0  -- manually verified by admin
);

CREATE TABLE omnicore_niche_profiles (
  id TEXT PRIMARY KEY,
  niche TEXT UNIQUE NOT NULL,
  sub_niches TEXT,        -- JSON array
  
  -- MARKET INTELLIGENCE
  avg_price_min REAL,
  avg_price_max REAL,
  avg_price_sweet_spot REAL,
  typical_margin_pct REAL,
  seasonal_peaks TEXT,    -- JSON: { month: multiplier }
  
  -- CONTENT PATTERNS
  top_keywords TEXT,      -- JSON array, sorted by conversion
  top_image_types TEXT,   -- JSON: { type: conversion_rate }
  best_cta_words TEXT,    -- JSON array
  tone_description TEXT,
  
  -- COMPETITIVE LANDSCAPE
  known_competitors TEXT, -- JSON array (anonymized)
  avg_competitor_count INTEGER,
  market_saturation TEXT, -- low | medium | high
  
  -- AI PATTERNS
  best_prompt_style TEXT,
  product_description_template TEXT,
  pricing_psychology TEXT,
  
  -- STATS
  stores_analyzed INTEGER DEFAULT 0,
  last_updated TEXT DEFAULT (datetime('now'))
);

CREATE TABLE omnicore_prompt_library (
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  description TEXT,
  niche TEXT,            -- NULL = universal
  use_case TEXT,         -- product_description | ab_variant | 
                         -- email_subject | image_brief | seo_title
  prompt_template TEXT NOT NULL,  -- with {{variables}}
  variables TEXT,        -- JSON: list of required variables
  avg_quality_score REAL DEFAULT 0,
  use_count INTEGER DEFAULT 0,
  created_by TEXT,       -- 'system' | admin user id
  is_public INTEGER DEFAULT 1,
  created_at TEXT DEFAULT (datetime('now'))
);

CREATE TABLE omnicore_search_cache (
  id TEXT PRIMARY KEY,
  query_hash TEXT UNIQUE,  -- hash of niche+query
  query TEXT,
  niche TEXT,
  results TEXT,            -- JSON cached results
  created_at TEXT DEFAULT (datetime('now')),
  expires_at TEXT          -- 7 days for most, 1 day for prices
);

async function learnFromEvent(event) {
  // Build memory object
  const memory = await buildMemoryFromEvent(event);
  
  // Check if similar memory exists (by niche + type + title similarity)
  const existing = await findSimilarMemory(memory);
  
  if (existing && similarity > 0.85) {
    // Update existing: increase confidence, update content
    await updateMemory(existing.id, {
      content: mergeMemories(existing.content, memory.content),
      confidence: Math.min(1.0, existing.confidence + 0.05),
      use_count: existing.use_count + 1,
      last_updated_at: new Date().toISOString()
    });
  } else {
    // Create new memory
    await createMemory(memory);
  }
  
  // Update niche profile aggregate
  await updateNicheProfile(event.niche, event);
}

PART 3: MEMORY RETRIEVAL — Smart Context Injection

This is the key: BEFORE every Claude call, retrieve relevant

MEMORY RETRIEVAL FUNCTION:
async function getRelevantMemories(niche, useCase, productType, limit=10) {
  return db.prepare(`
    SELECT * FROM omnicore_memories
    WHERE (niche = ? OR niche IS NULL OR niche = 'universal')
    AND (
      memory_type LIKE ? OR
      tags LIKE ?
    )
    AND confidence > 0.3
    ORDER BY 
      CASE WHEN niche = ? THEN 10 ELSE 0 END +  -- exact niche match
      CASE WHEN product_type = ? THEN 5 ELSE 0 END +  -- product match
      (confidence * 4) +  -- high confidence
      (success_rate * 3) +  -- proven success
      (use_count * 0.5)  -- frequently used
    DESC
    LIMIT ?
  `).all(niche, '%'+useCase+'%', '%'+useCase+'%', niche, productType, limit);
}

async function buildSystemPrompt(project, useCase) {
  const memories = await getRelevantMemories(
    project.niche, useCase, productType
  );
  const nicheProfile = await getNicheProfile(project.niche);
  
  const memoryContext = memories.length > 0 ? `
═══ OMNICORE MEMORY — Conocimiento acumulado para ${project.niche} ═══
${memories.map(m => `
[${m.memory_type.toUpperCase()}] ${m.title} (confianza: ${Math.round(m.confidence*100)}%)
${m.content}
`).join('\n')}
═══ FIN OMNICORE ═══
` : '';

  const nicheContext = nicheProfile ? `
PERFIL DEL NICHO "${project.niche}":
- Precio dulce del mercado: €${nicheProfile.avg_price_sweet_spot}
- Margen típico: ${nicheProfile.typical_margin_pct}%
- Top keywords que convierten: ${nicheProfile.top_keywords}
- Tipos de imagen con mejor conversión: ${nicheProfile.top_image_types}
- Tono que funciona en este nicho: ${nicheProfile.tone_description}
- Picos estacionales: ${nicheProfile.seasonal_peaks}
` : '';

  return `Eres ShopifyAI Expert...
${memoryContext}
${nicheContext}
Contexto tienda activa: ${project.name} | Nicho: ${project.niche}
`;
}

PART 4: OMNICORE SEARCH — Smarter than Google for eCommerce
═══════════════════════════════════════════════

When admin or AI needs to research a niche/product:
Instead of web search every time → check OmniCore first.

SEARCH FLOW:
1. Check omnicore_search_cache (hash of query+niche)
   If cache hit and not expired → return instantly (0ms)
   If cache miss → proceed

2. Check omnicore_memories for relevant stored knowledge
   If confidence > 0.7 memories exist → use them (10ms)
   
3. If gaps exist → do web search + Claude analysis
   Store results in OmniCore for future use

4. Return combined result: cached + new knowledge

PART 7: OMNICORE PRIVACY & ISOLATION
═══════════════════════════════════════════════

CRITICAL: Memory is anonymized and shared only at niche level.
Never at client level.

WHAT IS SHARED (anonymized):
✓ "In sports niche, lifestyle images convert better" (pattern)
✓ "Price sweet spot for running shoes: €89-119" (market data)
✓ "Keyword 'transpirable' converts in sports" (keyword insight)

WHAT IS NEVER SHARED:
✗ Client name, domain, revenue numbers
✗ Specific product titles or SKUs
✗ Client's pricing strategy
✗ Client's customer data
✗ Anything that could identify a specific store

MEMORY OWNERSHIP:
- Each client's RAW DATA belongs only to that client
- Only derived insights (anonymized patterns) feed OmniCore
- Client can opt-out of contributing to OmniCore (toggle in settings)
- Opting out = no cross-learning benefit (explain in UI)

GDPR COMPLIANCE:
- No personal data in memories
- All client data deletable without affecting OmniCore memories
- OmniCore memories are aggregate patterns, not personal data
- Privacy policy clearly explains the anonymized learning

// Middleware: inject OmniCore context into every Claude call
async function withOmniCore(projectId, useCase, claudeOptions) {
  const project = await getProject(projectId);
  const memories = await getRelevantMemories(
    project.niche, useCase, null, 8
  );
  const nicheProfile = await getNicheProfile(project.niche);
  
  // Inject into system prompt
  claudeOptions.system = buildSystemPrompt(
    project, memories, nicheProfile, claudeOptions.system
  );
  
  // Track that these memories were used
  await trackMemoryUsage(memories.map(m => m.id), projectId);
  
  return claudeOptions;
}

// Use in EVERY existing Claude call:
// Before:
const response = await callClaude({ system: basePrompt, ... });
// After:
const enhanced = await withOmniCore(projectId, 'product_description', 
  { system: basePrompt, ... });
const response = await callClaude(enhanced);
