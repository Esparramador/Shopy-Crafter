INVENTORY SYNC — CRON daily 7am:
For each active project:
1. GET /admin/api/2024-01/inventory_levels.json
2. GET /admin/api/2024-01/orders.json?status=any&limit=250
   (last 30 days to calculate avg_daily_sales)
3. Calculate: avg_daily_sales = units_sold_30d / 30
4. Calculate: days_remaining = current_stock / avg_daily_sales
5. Classify status:
   days_remaining > 45 AND avg_daily_sales < 0.1 → dead_stock
   days_remaining > 30 → healthy
   days_remaining 15-30 → warning
   days_remaining < 15 → critical
   current_stock = 0 → out_of_stock
6. If critical/out_of_stock AND days_remaining < supplier_lead_days:
   → Create restock alert + draft supplier email via Claude
   → Push notification to admin

VOICE ENGINE (JavaScript):
let recognition = null;

function initVoice() {
  if (!('webkitSpeechRecognition' in window) && 
      !('SpeechRecognition' in window)) {
    showToast('warning', 'Voz no disponible', 
      'Tu navegador no soporta reconocimiento de voz');
    return;
  }
  const SR = window.SpeechRecognition || window.webkitSpeechRecognition;
  recognition = new SR();
  recognition.lang = 'es-ES';
  recognition.continuous = false;
  recognition.interimResults = true;
  
  recognition.onresult = (e) => {
    const transcript = e.results[0][0].transcript;
    document.getElementById('vb-transcript').textContent = transcript;
    if (e.results[0].isFinal) processVoiceCommand(transcript);
  };
  
  recognition.onerror = (e) => {
    showToast('error', 'Error de voz', e.error);
    stopListening();
  };
}

function toggleVoice() {
  const btn = document.getElementById('voice-btn');
  if (btn.classList.contains('listening')) {
    stopListening();
  } else {
    startListening();
  }
}

function startListening() {
  document.getElementById('voice-btn').classList.add('listening');
  document.getElementById('voice-bubble').classList.remove('hidden');
  document.getElementById('vb-transcript').textContent = 'Escuchando...';
  recognition.start();
}

function stopListening() {
  document.getElementById('voice-btn').classList.remove('listening');
  recognition.stop();
}

async function processVoiceCommand(transcript) {
  stopListening();
  document.getElementById('vb-transcript').textContent = `"${transcript}"`;
  document.getElementById('vb-response').textContent = 'Procesando...';
  
  const currentProject = getCurrentProjectId(); // from app state
  
  const response = await fetch('/api/voice/command', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ 
      transcript, 
      projectId: currentProject,
      currentPage: window.location.pathname
    })
  });
  
  const { action, response: textResponse, executed } = await response.json();
  
  // Show text response
  document.getElementById('vb-response').textContent = textResponse;
  
  // Speak response
  if ('speechSynthesis' in window) {
    const utterance = new SpeechSynthesisUtterance(textResponse);
    utterance.lang = 'es-ES';
    utterance.rate = 1.0;
    utterance.pitch = 1.0;
    window.speechSynthesis.speak(utterance);
  }
  
  // Execute action if any
  if (action) executeVoiceAction(action);
  
  // Hide bubble after 5 seconds
  setTimeout(() => {
    document.getElementById('voice-bubble').classList.add('hidden');
  }, 5000);
}

  if (e.request.method !== 'GET') return;

// Push notifications handler

      data: { url: data.url || '/admin' },
      actions: data.actions || []
    }
  ));
});
