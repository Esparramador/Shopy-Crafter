STEP 1 — Create email template in Klaviyo:
POST /api/emails/push-to-klaviyo
Body: { flowId (your DB id) }

async function createKlaviyoTemplate(emailData) {
  const response = await fetch(
    `${KLAVIYO_BASE}/templates/`,
    {
      method: 'POST',
      headers: klaviyoHeaders,
      body: JSON.stringify({
        data: {
          type: 'template',
          attributes: {
            name: emailData.name,
            editor_type: 'CODE',
            html: emailData.html,
            text: emailData.text
          }
        }
      })
    }
  );
  const data = await response.json();
  return data.data.id; // klaviyo template id
}

STEP 2 — Create the flow in Klaviyo:
async function createKlaviyoFlow(flowConfig, templateId) {
  
  // Map your trigger names to Klaviyo metric names
  const TRIGGER_MAP = {
    'checkout_abandoned':     'Checkout Started',
    'order_placed':           'Placed Order', 
    'customer_subscribed':    'Subscribed to List',
    'review_requested':       'Fulfilled Order',
    'win_back':               'Placed Order',
    'vip_upgrade':            'Placed Order',
    'stock_back':             'Back in Stock',
    'price_drop':             'Viewed Product'
  };
  
  const response = await fetch(
    `${KLAVIYO_BASE}/flows/`,
    {
      method: 'POST', 
      headers: klaviyoHeaders,
      body: JSON.stringify({
        data: {
          type: 'flow',
          attributes: {
            name: flowConfig.name,
            status: 'live',  // activate immediately
            trigger_type: 'metric',
            trigger_options: {
              metric: {
                name: TRIGGER_MAP[flowConfig.triggerType]
              }
            }
          }
        }
      })
    }
  );
  const flowData = await response.json();
  return flowData.data.id;
}

STEP 3 — Add email action to flow:
async function addEmailToFlow(flowId, templateId, delay) {
  // Convert delay string to seconds
  const DELAY_MAP = {
    'immediate': 0,
    '1h': 3600,
    '3h': 10800, 
    '24h': 86400,
    '3d': 259200,
    '7d': 604800
  };
  
  await fetch(
    `${KLAVIYO_BASE}/flow-actions/`,
    {
      method: 'POST',
      headers: klaviyoHeaders, 
      body: JSON.stringify({
        data: {
          type: 'flow-action',
          attributes: {
            action_type: 'send_email',
            settings: {
              subject: '{{ subject }}', // set by template
              preview_text: '{{ preview_text }}',
              from_email: flowConfig.fromEmail,
              from_label: flowConfig.fromName,
              reply_to_email: flowConfig.replyTo
            },
            send_options: {
              use_smart_sending: true
            },
            rendering_options: {
              shorten_links: true,
              add_tracking_params: true,
              add_org_prefix: false
            }
          },
          relationships: {
            flow: { data: { type: 'flow', id: flowId } },
            template: { 
              data: { type: 'template', id: templateId } 
            }
          }
        }
      })
    }
  );
}

COMPLETE PUSH FUNCTION:
async function pushFlowToKlaviyo(flowId, projectId) {
  const flow = await getFlowFromDB(flowId);
  const project = await getProject(projectId);
  
  try {
    // 1. Create template
    const klaviyoTemplateId = await createKlaviyoTemplate({
      name: `[${project.name}] ${flow.name}`,
      html: flow.html_content,
      text: flow.text_content
    });
    
    // 2. Create flow
    const klaviyoFlowId = await createKlaviyoFlow({
      name: `[${project.name}] ${flow.name}`,
      triggerType: flow.trigger_type,
      fromEmail: project.from_email || 'noreply@tienda.com',
      fromName: project.name,
      replyTo: project.reply_email
    }, klaviyoTemplateId);
    
    // 3. Add email action with delay
    await addEmailToFlow(
      klaviyoFlowId, 
      klaviyoTemplateId,
      flow.send_delay
    );
    
    // 4. Save Klaviyo IDs to your DB
    await db.prepare(`
      UPDATE email_flows 
      SET klaviyo_template_id = ?,
          klaviyo_flow_id = ?,
          klaviyo_status = 'live',
          pushed_at = datetime('now')
      WHERE id = ?
    `).run(klaviyoTemplateId, klaviyoFlowId, flowId);
    
    return { success: true, klaviyoFlowId };
    
  } catch(err) {
    // Save error to DB
    await db.prepare(`
      UPDATE email_flows 
      SET klaviyo_status = 'error',
          klaviyo_error = ?
      WHERE id = ?
    `).run(err.message, flowId);
    throw err;
  }
}

PART 4: SYNC METRICS BACK FROM KLAVIYO
═══════════════════════════════════════════════

Pull email performance data to show in your app.
CRON: every 6 hours per active flow.

async function syncKlaviyoMetrics(klaviyoFlowId, flowId) {
  const response = await fetch(
    `${KLAVIYO_BASE}/flows/${klaviyoFlowId}/flow-actions/`,
    { headers: klaviyoHeaders }
  );
  const data = await response.json();
  
  // Extract metrics from flow actions
  const action = data.data[0];
  const stats = action?.attributes?.tracking_options || {};
  
  // Store in your DB
  await db.prepare(`
    UPDATE email_flows SET
      stat_sent = ?,
      stat_opened = ?,
      stat_clicked = ?,
      stat_open_rate = ?,
      stat_click_rate = ?,
      stat_revenue = ?,
      stats_updated_at = datetime('now')
    WHERE id = ?
  `).run(
    stats.sent || 0,
    stats.opened || 0, 
    stats.clicked || 0,
    stats.open_rate || 0,
    stats.click_rate || 0,
    stats.revenue || 0,
    flowId
  );
}

PART 6: DATABASE ADDITIONS
═══════════════════════════════════════════════

CREATE TABLE email_flows (
  id TEXT PRIMARY KEY,
  project_id TEXT NOT NULL,
  
  -- Your configuration
  name TEXT NOT NULL,
  flow_type TEXT NOT NULL,
  trigger_type TEXT NOT NULL,
  send_delay TEXT DEFAULT 'immediate',
  tone TEXT DEFAULT 'friendly',
  segment TEXT DEFAULT 'all',
  
  -- Generated content (by OmniCore)
  subject_a TEXT,
  subject_b TEXT,
  preview_text TEXT,
  html_content TEXT,
  text_content TEXT,
  generated_at TEXT,
  
  -- Klaviyo sync
  klaviyo_template_id TEXT,
  klaviyo_flow_id TEXT,
  klaviyo_status TEXT DEFAULT 'draft',
  -- draft | pushing | live | paused | error
  klaviyo_error TEXT,
  pushed_at TEXT,
  
  -- Metrics (synced from Klaviyo)
  stat_sent INTEGER DEFAULT 0,
  stat_opened INTEGER DEFAULT 0,
  stat_clicked INTEGER DEFAULT 0,
  stat_open_rate REAL DEFAULT 0,
  stat_click_rate REAL DEFAULT 0,
  stat_revenue REAL DEFAULT 0,
  stats_updated_at TEXT,
  
  is_active INTEGER DEFAULT 0,
  created_at TEXT DEFAULT (datetime('now')),
  updated_at TEXT DEFAULT (datetime('now'))
);

-- Email A/B test results
CREATE TABLE email_ab_results (
  id TEXT PRIMARY KEY,
  flow_id TEXT NOT NULL,
  variant TEXT NOT NULL, -- 'a' | 'b'
  subject TEXT,
  sent INTEGER DEFAULT 0,
  opened INTEGER DEFAULT 0,
  open_rate REAL DEFAULT 0,
  winner INTEGER DEFAULT 0,
  decided_at TEXT
);

-- Individual sends log (optional, for debugging)
CREATE TABLE email_send_log (
  id TEXT PRIMARY KEY,
  flow_id TEXT,
  project_id TEXT,
  event TEXT, -- sent|opened|clicked|bounced
  klaviyo_message_id TEXT,
  created_at TEXT DEFAULT (datetime('now'))
);
