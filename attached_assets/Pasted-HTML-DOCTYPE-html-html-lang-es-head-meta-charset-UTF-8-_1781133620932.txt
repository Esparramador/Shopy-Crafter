    <script>
        function nextStep(currentStepNum) {
            // Buscamos el paso actual y el siguiente
            const currentStep = document.getElementById(`step-${currentStepNum}`);
            const nextStep = document.getElementById(`step-${currentStepNum + 1}`);

            // Validar que el input no esté vacío (opcional)
            const input = currentStep.querySelector('input');
            if(input && input.value.trim() === '') {
                alert('Por favor, completa este campo.');
                input.focus();
                return;
            }

            // Animación: Quitamos la clase 'active' al actual y le ponemos 'past' (se va hacia arriba)
            currentStep.classList.remove('active');
            currentStep.classList.add('past');

            // Animación: Le ponemos la clase 'active' al siguiente (entra desde abajo)
            if (nextStep) {
                nextStep.classList.add('active');
                
                // Hacer auto-focus en el siguiente input después de la animación
                setTimeout(() => {
                    const nextInput = nextStep.querySelector('input');
                    if(nextInput) nextInput.focus();
                }, 600); // 600ms es lo que dura la transición CSS
            }
        }
    </script>
