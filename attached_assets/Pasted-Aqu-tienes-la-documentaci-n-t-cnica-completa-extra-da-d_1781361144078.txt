🎮 Paso 2: El Controlador de Estados (JavaScript)
Vamos a crear el objeto player que imita la máquina de estados de Visme (Entrance -> Waiting -> Ending).

JavaScript
import * as THREE from 'three';
import { GLTFLoader } from 'three/examples/jsm/loaders/GLTFLoader.js';

class InteractivePlayer {
  constructor() {
    this.mixer = null;
    this.actions = {}; // Aquí guardaremos { entrance, waiting, ending }
    this.activeAction = null;
    
    this.initScene();
  }

  initScene() {
    // 1. Configuración básica de Three.js
    const canvas = document.getElementById('webgl-canvas');
    this.renderer = new THREE.WebGLRenderer({ canvas, alpha: true, antialias: true });
    this.scene = new THREE.Scene();
    this.camera = new THREE.PerspectiveCamera(45, 800/600, 0.1, 100);
    this.camera.position.set(0, 1.5, 5);

    // 2. Grupos de Control (replicando la estructura original)
    this.visibilityController = new THREE.Group();
    this.scene.add(this.visibilityController);

    // Luces
    const light = new THREE.DirectionalLight(0xffffff, 1);
    light.position.set(0, 2, 2);
    this.scene.add(light, new THREE.AmbientLight(0xffffff, 0.5));

    this.loadCharacter();
  }

  loadCharacter() {
    const loader = new GLTFLoader();
    // En un entorno real, cargarías tu propio modelo GLB con las 3 animaciones
    loader.load('ruta/a/tu/personaje_con_animaciones.glb', (gltf) => {
      this.visibilityController.add(gltf.scene);
      this.mixer = new THREE.AnimationMixer(gltf.scene);

      // Asumiendo que tu GLB tiene las animaciones en este orden:
      this.actions.entrance = this.mixer.clipAction(gltf.animations[0]);
      this.actions.waiting = this.mixer.clipAction(gltf.animations[1]);
      this.actions.ending = this.mixer.clipAction(gltf.animations[2]);

      // Configurar bucles según la documentación
      this.actions.entrance.setLoop(THREE.LoopOnce);
      this.actions.waiting.setLoop(THREE.LoopRepeat);
      this.actions.ending.setLoop(THREE.LoopOnce);

      // Cuando termine la entrada, pasar a modo espera automáticamente
      this.mixer.addEventListener('finished', (e) => {
        if (e.action === this.actions.entrance) {
          this.playWaiting();
        }
      });

      // Iniciar la secuencia
      this.playEntrance();
    });
  }

  // --- MÉTODOS DE ETAPAS ---

  playEntrance() {
    if (this.activeAction) this.activeAction.stop();
    this.activeAction = this.actions.entrance;
    this.activeAction.reset().play();
    
    // Animación CSS del formulario entrando
    document.querySelector('.formContainer').style.transform = 'matrix(1, 0, 0, 1, 0, 0)';
  }

  playWaiting() {
    // Transición suave entre animaciones (crossfade)
    this.actions.waiting.reset().play();
    this.actions.entrance.crossFadeTo(this.actions.waiting, 0.5, false);
    this.activeAction = this.actions.waiting;
  }

  playEnding() {
    this.actions.ending.reset().play();
    this.actions.waiting.crossFadeTo(this.actions.ending, 0.2, false);
    this.activeAction = this.actions.ending;
  }
}

// Inicializar
const player = new InteractivePlayer();

// Conectar con el DOM
document.getElementById('submitBtn').addEventListener('click', () => {
  player.playEnding(); // Disparar la celebración
});
💡 Cómo encaja todo
La Entrada: Al cargar la página, player.playEntrance() reproduce la animación 3D de entrada (LoopOnce).

El Enlace (Event Listener): El evento finished del AnimationMixer detecta que la entrada terminó e inmediatamente dispara player.playWaiting().

El Estado Inactivo: El personaje se queda respirando o mirando el reloj (LoopRepeat) gracias a la segunda animación.

La Interacción: Cuando el usuario hace clic en el botón HTML de enviar, el DOM le avisa a WebGL llamando a player.playEnding(), haciendo que el personaje celebre.

3. Integración de la Mano 3D (Three.js)
Este script asume que cuentas con un archivo .glb de una mano realista (con un esqueleto posicionado para señalar). Configura la cámara para emular la vista de ángulo inferior-derecha.

JavaScript
import * as THREE from 'three';
import { GLTFLoader } from 'three/examples/jsm/loaders/GLTFLoader.js';

const init3DHand = () => {
  const canvas = document.getElementById('hand-gesture-canvas');
  // Ajustado a las dimensiones del landscape 900x473
  const renderer = new THREE.WebGLRenderer({ canvas, alpha: true, antialias: true });
  renderer.setSize(900, 473);

  const scene = new THREE.Scene();
  
  // Iluminación suave para resaltar el tono cálido (#C68642) y las sombras (#8B5A2B)
  const ambientLight = new THREE.AmbientLight(0xffffff, 0.8);
  const directionalLight = new THREE.DirectionalLight(0xffe6cc, 1.2); // Luz ligeramente cálida
  directionalLight.position.set(2, 3, 4);
  scene.add(ambientLight, directionalLight);

  // Cámara posicionada para la esquina inferior derecha
  const camera = new THREE.PerspectiveCamera(45, 900/473, 0.1, 100);
  camera.position.set(3, -1, 5); // Vista lateral en ángulo

  const loader = new GLTFLoader();
  
  loader.load('assets/3d/hand_pointing.glb', (gltf) => {
    const handModel = gltf.scene;
    
    // Posicionamiento de la mano: Esquina inferior derecha, apuntando al botón centrado
    handModel.position.set(2.5, -2, 0); 
    handModel.rotation.set(0, -Math.PI / 4, Math.PI / 6); // Ángulo de ~45° apuntando arriba-izquierda
    
    // Escala: ~25-30% del alto del canvas
    handModel.scale.set(1.5, 1.5, 1.5); 
    
    scene.add(handModel);

    // Bucle de renderizado simple (aquí iría el AnimationMixer si la mano está animada)
    const animate = () => {
      requestAnimationFrame(animate);
      renderer.render(scene, camera);
    };
    animate();
  });
};

Integrar React Toastify en esta arquitectura es una decisión excelente. En lugar de limitarnos a un simple "check verde" estático dentro del input (como en la Variante 2 original de Visme), podemos usar Toastify para manejar las notificaciones de éxito y error con animaciones fluidas que complementen la entrada del personaje 3D, manteniendo la estética fotorrealista y profesional.

Aquí tienes la implementación en React que fusiona tu diseño base en CSS, la máscara del canvas 3D y las notificaciones de Toastify personalizadas para la paleta de la Agencia IA.

1. El Componente React (Formulario + 3D + Toastify)
Este componente maneja el estado del formulario y dispara la notificación estilizada al hacer clic en el CTA.

JavaScript
import React, { useState, useEffect, useRef } from 'react';
import { ToastContainer, toast } from 'react-toastify';
import 'react-toastify/dist/ReactToastify.css';
import * as THREE from 'three';
import { GLTFLoader } from 'three/examples/jsm/loaders/GLTFLoader';
import './AgencyForm.css';

export default function AgencyForm() {
  const [email, setEmail] = useState('');
  const [isSubmitted, setIsSubmitted] = useState(false);
  const canvasRef = useRef(null);

  // Inicialización de WebGL/Three.js
  useEffect(() => {
    if (!canvasRef.current) return;

    const renderer = new THREE.WebGLRenderer({ canvas: canvasRef.current, alpha: true, antialias: true });
    renderer.setSize(900, 473);
    const scene = new THREE.Scene();
    
    // Iluminación para tono cálido fotorrealista (#C68642)
    const ambientLight = new THREE.AmbientLight(0xffffff, 0.8);
    const dirLight = new THREE.DirectionalLight(0xffe6cc, 1.2);
    dirLight.position.set(2, 3, 4);
    scene.add(ambientLight, dirLight);

    const camera = new THREE.PerspectiveCamera(45, 900 / 473, 0.1, 100);
    camera.position.set(3, -1, 5);

    // Carga de la Mano 3D (Señalando al botón)
    const loader = new GLTFLoader();
    loader.load('/assets/3d/hand_pointing.glb', (gltf) => {
      const hand = gltf.scene;
      hand.position.set(2.5, -2, 0); 
      hand.rotation.set(0, -Math.PI / 4, Math.PI / 6);
      hand.scale.set(1.5, 1.5, 1.5);
      scene.add(hand);

      const animate = () => {
        requestAnimationFrame(animate);
        renderer.render(scene, camera);
      };
      animate();
    });

    return () => renderer.dispose();
  }, []);

  const handleSubmit = (e) => {
    e.preventDefault();
    if (!email) {
      toast.error('Por favor, ingresa un correo válido.', { theme: 'colored' });
      return;
    }

    // Disparar React Toastify con el estilo de la agencia
    toast.success('¡Inscripción confirmada! Te contactaremos pronto.', {
      icon: "✨",
      className: 'agency-toast-success',
      progressClassName: 'agency-toast-progress',
    });

    setIsSubmitted(true);
  };

  return (
    <div className="ia-event-container">
      {/* CAPA 2: EFECTOS (Ray Lines) */}
      <div className="starburst-effects"></div>

      {/* CAPA 5: CHARACTER (Mano 3D en WebGL superpuesta) */}
      <div className="canvas-mask">
        <canvas ref={canvasRef}></canvas>
      </div>

      {/* CAPA 3: CLASSIC CARD */}
      <div className="classic-card">
        <h2 className="card-title">Solicita tu Demo de IA</h2>
        <p className="card-subtitle">Descubre cómo automatizar tus procesos</p>

        <form onSubmit={handleSubmit} className="form-content">
          <div className="field-placeholder">Nombre</div>
          <div className="field-placeholder">Empresa</div>
          
          <div className={`field-active ${isSubmitted ? 'success-state' : ''}`}>
            <input 
              type="email" 
              placeholder="joan@agencia-ia.com" 
              value={email}
              onChange={(e) => setEmail(e.target.value)}
              disabled={isSubmitted}
            />
            {isSubmitted && (
              <span className="check-icon">
                <svg viewBox="0 0 24 24" width="14" height="14" fill="#4CAF50">
                  <path d="M9 16.17L4.83 12l-1.42 1.41L9 19 21 7l-1.41-1.41z"/>
                </svg>
              </span>
            )}
          </div>

          {/* CAPA 4: BOTÓN SUBMIT */}
          <button type="submit" className="submit-button" disabled={isSubmitted}>
            {isSubmitted ? 'ENVIADO' : 'QUIERO MI DEMO'}
          </button>
        </form>
      </div>

      {/* Contenedor de notificaciones */}
      <ToastContainer position="bottom-center" autoClose={4000} hideProgressBar={false} />
    </div>
  );
}

La Ventaja de esta Arquitectura
Al separar la lógica del renderizado 3D en un useEffect y enmascararlo con pointer-events: none en la clase .canvas-mask, logramos que el usuario interactúe limpiamente con los inputs HTML nativos y los botones. Cuando se dispara el evento handleSubmit, react-toastify inyecta la notificación en el DOM superior (z-index: 9999), sobreponiéndose elegantemente al modelo 3D y a la tarjeta.

Si tuviéramos acceso a múltiples animaciones dentro del archivo .glb (por ejemplo, el gesto de "pulgar arriba" que mencionaste en tu Variante 3), podríamos hacer que el modelo 3D cambie de postura exactamente en el momento en que toast.success se dispara en pantalla.
