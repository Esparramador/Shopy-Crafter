const STOREFRONT_ENDPOINT = 
  `https://${process.env.SHOP_DOMAIN}/api/2024-10/graphql.json`;

const PRODUCTS_QUERY = `
  query getCollectionProducts($handle: String!) {
    collectionByHandle(handle: $handle) {
      title
      products(first: 20) {
        edges {
          node {
            id
            title
            handle
            descriptionHtml
            productType
            tags
            priceRange {
              minVariantPrice {
                amount
                currencyCode
              }
            }
            compareAtPriceRange {
              minVariantPrice {
                amount
                currencyCode
              }
            }
            images(first: 3) {
              edges {
                node {
                  url
                  altText
                  width
                  height
                }
              }
            }
            variants(first: 5) {
              edges {
                node {
                  id
                  title
                  price {
                    amount
                    currencyCode
                  }
                  compareAtPrice {
                    amount
                    currencyCode
                  }
                  availableForSale
                  selectedOptions {
                    name
                    value
                  }
                }
              }
            }
            metafields(
              identifiers: [
                {namespace: "custom", key: "features"},
                {namespace: "custom", key: "highlight"},
                {namespace: "custom", key: "badge"}
              ]
            ) {
              key
              value
            }
          }
        }
      }
    }
  }
`;

router.get('/api/store/products', async (req, res) => {
  try {
    const response = await fetch(STOREFRONT_ENDPOINT, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'X-Shopify-Storefront-Access-Token': 
          process.env.STOREFRONT_ACCESS_TOKEN
      },
      body: JSON.stringify({
        query: PRODUCTS_QUERY,
        variables: { 
          handle: process.env.COLLECTION_HANDLE 
        }
      })
    });
    
    const data = await response.json();
    const products = data.data?.collectionByHandle
      ?.products?.edges?.map(e => e.node) || [];
    
    // Cache for 30 minutes — products don't change often
    res.set('Cache-Control', 'public, max-age=1800');
    res.json({ products, currency: process.env.SHOP_CURRENCY });
    
  } catch(err) {
    res.status(500).json({ error: 'Store temporarily unavailable' });
  }
});

POST /api/store/checkout
No auth required — public endpoint.
Creates a Shopify checkout and returns the URL.

const CHECKOUT_QUERY = `
  mutation cartCreate($lines: [CartLineInput!]!) {
    cartCreate(input: { lines: $lines }) {
      cart {
        id
        checkoutUrl
      }
      userErrors {
        field
        message
      }
    }
  }
`;

router.post('/api/store/checkout', async (req, res) => {
  const { variantId, quantity = 1 } = req.body;
  
  try {
    const response = await fetch(STOREFRONT_ENDPOINT, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'X-Shopify-Storefront-Access-Token':
          process.env.STOREFRONT_ACCESS_TOKEN
      },
      body: JSON.stringify({
        query: CHECKOUT_QUERY,
        variables: {
          lines: [{ merchandiseId: variantId, quantity }]
        }
      })
    });
    
    const data = await response.json();
    const checkoutUrl = data.data?.cartCreate?.cart?.checkoutUrl;
    
    if (!checkoutUrl) throw new Error('Checkout failed');
    
    // Return URL — frontend redirects to Shopify checkout
    res.json({ checkoutUrl });
    
  } catch(err) {
    res.status(500).json({ error: 'Could not create checkout' });
  }
});

function renderProductCard(product) {
  const price = parseFloat(
    product.priceRange.minVariantPrice.amount
  );
  const compareAt = parseFloat(
    product.compareAtPriceRange?.minVariantPrice?.amount || 0
  );
  const variant = product.variants.edges[0]?.node;
  const image = product.images.edges[0]?.node;
  
  // Get metafields
  const featuresField = product.metafields
    .find(m => m?.key === 'features');
  const features = featuresField 
    ? JSON.parse(featuresField.value) 
    : [];
  const badge = product.metafields
    .find(m => m?.key === 'badge')?.value;
  const highlight = product.metafields
    .find(m => m?.key === 'highlight')?.value;
  
  const discount = compareAt > price 
    ? Math.round((1 - price/compareAt) * 100) 
    : 0;
  
  const isSubscription = product.productType === 'subscription'
    || product.tags.includes('subscription');
  
  return `
    <div class="product-card ${highlight ? 'featured' : ''}">
      ${badge ? `<div class="product-badge">${badge}</div>` : ''}
      
      ${image ? `
        <div class="product-image">
          <img src="${image.url}" 
               alt="${image.altText || product.title}"
               loading="lazy">
        </div>` : `
        <div class="product-image product-image-placeholder">
          <div class="pip-icon">⚡</div>
        </div>`
      }
      
      <div class="product-body">
        <div class="product-type-tag">
          ${isSubscription ? 'Servicio mensual' : 'Pago único'}
        </div>
        
        <h3 class="product-title">${product.title}</h3>
        
        <div class="product-price-row">
          <div class="product-price">
            €${price % 1 === 0 ? price : price.toFixed(2)}
            ${isSubscription ? '<span>/mes</span>' : ''}
          </div>
          ${compareAt > price ? `
            <div class="product-compare">
              €${compareAt % 1 === 0 
                 ? compareAt 
                 : compareAt.toFixed(2)}
            </div>
            <div class="product-discount">-${discount}%</div>
          ` : ''}
        </div>
        
        ${features.length > 0 ? `
          <ul class="product-features">
            ${features.slice(0, 5).map(f => `
              <li>
                <span class="pf-check">✓</span>
                ${f}
              </li>
            `).join('')}
          </ul>
        ` : `
          <div class="product-desc">
            ${product.descriptionHtml.substring(0, 150)}...
          </div>
        `}
        
        <button 
          class="btn ${highlight ? 'btn-gold' : 'btn-ghost'} 
                 product-cta"
          onclick="buyProduct('${variant?.id}', '${product.title}')"
          ${!variant?.availableForSale ? 'disabled' : ''}>
          ${!variant?.availableForSale 
            ? 'No disponible' 
            : isSubscription 
              ? 'Suscribirse →' 
              : 'Adquirir →'}
        </button>
      </div>
    </div>
  `;
}

async function loadStore() {
  try {
    const res = await fetch('/api/store/products');
    const { products } = await res.json();
    
    // Separate one-time from subscriptions
    const oneTime = products.filter(p => 
      p.productType !== 'subscription' && 
      !p.tags.includes('subscription')
    );
    const subscriptions = products.filter(p =>
      p.productType === 'subscription' || 
      p.tags.includes('subscription')
    );
    
    // Render one-time products
    const grid = document.getElementById('storeGrid');
    grid.innerHTML = oneTime.map(renderProductCard).join('');
    
    // Render subscriptions if any
    if (subscriptions.length > 0) {
      const mGrid = document.getElementById('maintenanceGrid');
      mGrid.innerHTML = subscriptions
        .map(renderProductCard).join('');
      document.getElementById('storeMaintenance')
        .style.display = 'block';
    }
    
    // Show grid, hide loading
    document.getElementById('storeLoading').style.display = 'none';
    grid.style.display = 'grid';
    
  } catch(err) {
    document.getElementById('storeLoading').innerHTML = `
      <div class="empty-state">
        <div class="empty-icon">🛍️</div>
        <div class="empty-title">Tienda no disponible</div>
        <div class="empty-desc">
          Contacta directamente para adquirir tu pack.
        </div>
        <a href="mailto:hola@shopifyai.pro" 
           class="btn btn-gold">Contactar →</a>
      </div>`;
  }
}

async function buyProduct(variantId, productName) {
  // Show loading on button
  const btn = event.target;
  btn.classList.add('loading');
  btn.disabled = true;
  
  try {
    const res = await fetch('/api/store/checkout', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ variantId, quantity: 1 })
    });
    
    const { checkoutUrl, error } = await res.json();
    
    if (error || !checkoutUrl) throw new Error(error);
    
    // Track the click (optional analytics)
    console.log(`Checkout started: ${productName}`);
    
    // Redirect to Shopify checkout
    // Opens in same tab — Shopify handles everything
    window.location.href = checkoutUrl;
    
  } catch(err) {
    btn.classList.remove('loading');
    btn.disabled = false;
    
    // Fallback: direct link to store
    showToast('info', 
      'Redirigiendo a la tienda...', 
      'Serás redirigido a la página de pago');
    
    setTimeout(() => {
      window.location.href = 
        `https://${process.env.SHOP_DOMAIN_PUBLIC}/collections/shopifyai-pro`;
    }, 1500);
  }
}

// Load store when page loads
document.addEventListener('DOMContentLoaded', loadStore);
