typescriptimport { Router } from 'express';
import { requireAuth } from '../middleware/auth';
import Anthropic from '@anthropic-ai/sdk';

const router = Router();

// POST /api/suppliers/search
router.post('/search', requireAuth, async (req, res) => {
  const { query, sources, category } = req.body;
  const { shopId } = req.user;

  // 1. Crear registro de búsqueda
  const search = await db.insert(supplierSearches).values({
    shopId, query, sources, status: 'running'
  }).returning();

  // 2. Lanzar búsquedas en paralelo
  const results = await Promise.all([
    sources.includes('google') && searchGoogle(query),
    sources.includes('bing') && searchBing(query),
    sources.includes('duckduckgo') && searchDuckDuckGo(query),
    sources.includes('instagram') && searchInstagram(query),
    sources.includes('shopify') && searchShopifyStores(query),
  ].filter(Boolean));

  // 3. Combinar y deduplicar URLs
  const allUrls = [...new Set(results.flat().map(r => r.url))];

  // 4. Enriquecer con Claude (extracción estructurada)
  const enriched = await Promise.all(
    allUrls.slice(0, 50).map(url => extractSupplierData(url, category))
  );

  // 5. Guardar nuevos proveedores (deduplicar por website/cif)
  const newSuppliers = await dedupeAndInsert(enriched, shopId);

  res.json({
    searchId: search[0].id,
    totalFound: enriched.length,
    newSuppliers: newSuppliers.length,
    suppliers: newSuppliers
  });
});

🧮 Motor de cálculo COGS (/server/routes/cogs-engine.ts)
typescript// Sigue tu metodología: ASSUMPTIONS → CÁLCULOS → RESUMEN

interface COGSInputs {
  supplierId: number;
  unitPrice: number;
  volume: number;
  transportPerUnit?: number;
  specialTaxPerUnit?: number;
  breakageRatePct?: number;     // 3-8% para vinos
  coldStorageCostPerUnit?: number;
  staffCostPerUnit?: number;
  marketingCostPerUnit?: number;
  formationCostPerUnit?: number;
}

function calculateCOGS(inputs: COGSInputs) {
  const {
    unitPrice, volume,
    transportPerUnit = 0,
    specialTaxPerUnit = 0,
    breakageRatePct = 5,
    coldStorageCostPerUnit = 0,
    staffCostPerUnit = 0,
    marketingCostPerUnit = 0,
    formationCostPerUnit = 0,
  } = inputs;

  // COGS Unitario = Precio + Transporte + Impuestos + Rotura + Frío + Personal + Marketing
  const breakageCost = unitPrice * (breakageRatePct / 100);
  
  const cogsUnit = 
    unitPrice + 
    transportPerUnit + 
    specialTaxPerUnit + 
    breakageCost + 
    coldStorageCostPerUnit + 
    staffCostPerUnit + 
    marketingCostPerUnit + 
    formationCostPerUnit;

  // Rangos Bajo/Medio/Alto
  return {
    cogsLow: cogsUnit * 0.92,
    cogsAvg: cogsUnit,
    cogsHigh: cogsUnit * 1.08,
    cogsTotal: cogsUnit * volume,
    breakdown: {
      price: unitPrice,
      transport: transportPerUnit,
      tax: specialTaxPerUnit,
      breakage: breakageCost,
      cold: coldStorageCostPerUnit,
      staff: staffCostPerUnit,
      marketing: marketingCostPerUnit,
    }
  };
}

// Endpoint comparativa entre proveedores
router.post('/compare', requireAuth, async (req, res) => {
  const { supplierIds, productName, volume } = req.body;
  
  const comparisons = await Promise.all(
    supplierIds.map(async (id) => {
      const pricing = await db.query.supplierPricing.findFirst({
        where: and(
          eq(supplierPricing.supplierId, id),
          eq(supplierPricing.productName, productName)
        )
      });
      
      const cogs = calculateCOGS({
        supplierId: id,
        unitPrice: Number(pricing.unitPrice),
        volume,
        // ... resto de costes
      });
      
      return { supplierId: id, ...cogs };
    })
  );

  // Ranking por COGS
  const ranked = comparisons.sort((a, b) => a.cogsAvg - b.cogsAvg);
  const best = ranked[0];
  const worst = ranked[ranked.length - 1];
  const savingsAnnual = (worst.cogsAvg - best.cogsAvg) * volume;

  res.json({ comparisons: ranked, best, savingsAnnual });
});
