// Excel

    .addInputCell('B6', 0.45, { color: 'yellow', label: 'Transporte' })
  .addSheet('Cálculos')
    .addFormulaCell('B5', '=Assumptions!B5+Assumptions!B6+Assumptions!B7')

    .addKPICard('B2', 'Margen', '68%', { color: 'teal', large: true })

await excel.saveTo('/path/to/file.xlsx');

// PPTX (mismo estilo)
const pptx = new PPTXBuilder()

  .addCoverSlide({ title: 'Análisis Q1 2026', subtitle: 'Hotel Mediterráneo' })

  .addBarChartSlide({ title: 'Comparativa proveedores', data: [...] })
  .addTableSlide({ title: 'Top 5', rows: [...] })
  .addPlan90DaysSlide({ phases: [...] })
  .build();

// DOCX
const docx = new DOCXBuilder()

  .addTitle('Contrato Suministro 2026')
  .addParagraph('...')
  .addTable({ headers: [...], rows: [...] })

  .build();

// PDF (vía HTML)
const pdf = new PDFBuilder()

- Validación de inputs (lanza error si datos malformados)

PATRÓN ESTÁNDAR DE CADA ENDPOINT:

1. Valida payload con Zod
2. Carga contexto necesario (producto, shop, branding)

4. Llama al Builder correspondiente
5. Guarda archivo en /storage/reports/{shopId}/{reportId}.{ext}
6. Inserta registro en tabla generated_reports
7. Devuelve { reportId, downloadUrl, previewUrl, generatedAt }

Schema necesario:

```sql
CREATE TABLE generated_reports (
  id TEXT PRIMARY KEY,
  shop_id INTEGER NOT NULL,
  user_id INTEGER,
  type TEXT NOT NULL,         -- 'excel-cogs', 'pptx-executive', etc
  format TEXT NOT NULL,       -- 'xlsx', 'pptx', 'docx', 'pdf', 'html'
  filename TEXT NOT NULL,
  file_path TEXT NOT NULL,
  file_size INTEGER,
  ai_cost NUMERIC,            -- coste de las llamadas IA
  generation_time_ms INTEGER,
  context JSONB,              -- payload original
  created_at TIMESTAMP DEFAULT NOW(),
  expires_at TIMESTAMP        -- los archivos se borran a los 30 días
);
```
