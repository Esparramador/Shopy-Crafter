import type {
  ABTest,
  ABTestingKPIs,
  TestHistoryEntry,
  ShopifyProduct,
  ImageAnalysis,
  ImageVariant,
  ImageStyle,
  CompetitorAnalysis,
  SupplierImpactAnalysis,
  ImageTestConfig,
  PriceTestConfig,
  TestStats,
} from './types';

const API_BASE = '/api/ab-testing';

async function request<T>(url: string, options?: RequestInit): Promise<T> {
  const response = await fetch(url, {
    ...options,
    credentials: 'include',
    headers: { 'Content-Type': 'application/json', ...options?.headers },
  });
  if (!response.ok) {
    let msg = `Error ${response.status}`;
    try {
      const data = await response.json();
      msg = data.message || data.error || msg;
    } catch { /* noop */ }
    throw new Error(msg);
  }
  return response.json();
}

export const abTestingAPI = {
  // ============================================================
  // KPIs Y LISTA
  // ============================================================

  getKPIs: (): Promise<ABTestingKPIs> =>
    request(`${API_BASE}/kpis`),

  getActiveTests: (): Promise<ABTest[]> =>
    request(`${API_BASE}/tests/active`),

  getHistory: (limit = 50): Promise<TestHistoryEntry[]> =>
    request(`${API_BASE}/tests/history?limit=${limit}`),

  getTest: (testId: string): Promise<ABTest> =>
    request(`${API_BASE}/tests/${testId}`),

  // ============================================================
  // PRODUCTOS (vinculados a Shopify)
  // ============================================================

  getProducts: (): Promise<ShopifyProduct[]> =>
    request(`${API_BASE}/products`),

  // ============================================================
  // ANÁLISIS DE IMAGEN ACTUAL (Claude analiza la imagen del producto)
  // ============================================================

  /**
   * Analiza la imagen actual del producto Shopify con IA.
   * Devuelve scores, debilidades y sugerencias de mejora.
   */
  analyzeImage: (productId: string, imageUrl: string): Promise<ImageAnalysis> =>
    request(`${API_BASE}/image/analyze`, {
      method: 'POST',
      body: JSON.stringify({ productId, imageUrl }),
    }),

  // ============================================================
  // GENERACIÓN DE VARIANTES IA
  // ============================================================

  /**
   * Genera UNA variante de imagen con un estilo específico.
   * Usa Runway/Replicate/etc. según disponibilidad.
   */
  generateImageVariant: (
    productId: string,
    style: ImageStyle,
    customPrompt?: string
  ): Promise<ImageVariant> =>
    request(`${API_BASE}/image/generate-variant`, {
      method: 'POST',
      body: JSON.stringify({ productId, style, customPrompt }),
    }),

  /**
   * Genera 4 variantes en paralelo (lifestyle, studio, context, detail).
   */
  generateAllVariants: (productId: string): Promise<ImageVariant[]> =>
    request(`${API_BASE}/image/generate-all`, {
      method: 'POST',
      body: JSON.stringify({ productId }),
    }),

  // ============================================================
  // ANÁLISIS DE COMPETENCIA (scraping multi-fuente)
  // ============================================================

  /**
   * Busca el producto en múltiples fuentes (Amazon, Shopify stores,
   * marketplaces) y devuelve precios scrapeados.
   */
  analyzeCompetitors: (productId: string): Promise<CompetitorAnalysis> =>
    request(`${API_BASE}/price/competitors`, {
      method: 'POST',
      body: JSON.stringify({ productId }),
    }),

  // ============================================================
  // ANÁLISIS DE PROVEEDORES (conecta con Supplier Intelligence)
  // ============================================================

  /**
   * Busca proveedores alternativos para reducir COGS.
   * Conecta con el módulo Supplier Intelligence.
   */
  analyzeSupplierImpact: (productId: string): Promise<SupplierImpactAnalysis> =>
    request(`${API_BASE}/price/supplier-impact`, {
      method: 'POST',
      body: JSON.stringify({ productId }),
    }),

  // ============================================================
  // RECOMENDACIÓN DE PRECIO (combina competencia + proveedores + COGS)
  // ============================================================

  /**
   * Claude analiza competencia + COGS + proveedores y devuelve
   * 3 niveles de precio (conservador / óptimo / agresivo).
   */
  recommendPrice: (productId: string): Promise<{
    conservative: number;
    optimal: number;
    aggressive: number;
    justification: string;
    risks: string[];
  }> =>
    request(`${API_BASE}/price/recommend`, {
      method: 'POST',
      body: JSON.stringify({ productId }),
    }),

  // ============================================================
  // CREACIÓN Y CONTROL DE TESTS
  // ============================================================

  createImageTest: (config: ImageTestConfig): Promise<ABTest> =>
    request(`${API_BASE}/tests/image`, {
      method: 'POST',
      body: JSON.stringify(config),
    }),

  createPriceTest: (config: PriceTestConfig): Promise<ABTest> =>
    request(`${API_BASE}/tests/price`, {
      method: 'POST',
      body: JSON.stringify(config),
    }),

  startTest: (testId: string): Promise<ABTest> =>
    request(`${API_BASE}/tests/${testId}/start`, { method: 'POST' }),

  pauseTest: (testId: string): Promise<ABTest> =>
    request(`${API_BASE}/tests/${testId}/pause`, { method: 'POST' }),

  resumeTest: (testId: string): Promise<ABTest> =>
    request(`${API_BASE}/tests/${testId}/resume`, { method: 'POST' }),

  cancelTest: (testId: string): Promise<{ cancelled: boolean }> =>
    request(`${API_BASE}/tests/${testId}/cancel`, { method: 'POST' }),

  /**
   * Polling de stats en vivo del test.
   */
  getTestStats: (testId: string): Promise<TestStats> =>
    request(`${API_BASE}/tests/${testId}/stats`),

  /**
   * Declara ganador y aplica la variante a Shopify.
   */
  declareWinner: (testId: string, winnerVariant: 'A' | 'B'): Promise<{
    applied: boolean;
    insights: string[];
  }> =>
    request(`${API_BASE}/tests/${testId}/declare-winner`, {
      method: 'POST',
      body: JSON.stringify({ winnerVariant }),
    }),

  /**
   * Genera reporte PDF del test completado.
   */
  generateReport: (testId: string): Promise<{ reportUrl: string }> =>
    request(`${API_BASE}/tests/${testId}/report`, { method: 'POST' }),
};
