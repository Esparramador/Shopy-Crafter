ENCRYPTION for credentials:
Use AES-256-GCM via Node.js crypto module.
Master key stored in .env as ENCRYPTION_KEY (32-byte hex).
Never store raw tokens — always encrypt before saving, decrypt on use.

const crypto = require('crypto');
const ALGO = 'aes-256-gcm';
const KEY = Buffer.from(process.env.ENCRYPTION_KEY, 'hex');

function encrypt(text) {
  const iv = crypto.randomBytes(16);
  const cipher = crypto.createCipheriv(ALGO, KEY, iv);
  const encrypted = Buffer.concat([cipher.update(text,'utf8'), cipher.final()]);
  const tag = cipher.getAuthTag();
  return iv.toString('hex')+':'+tag.toString('hex')+':'+encrypted.toString('hex');
}

function decrypt(data) {
  const [ivHex, tagHex, encHex] = data.split(':');
  const iv = Buffer.from(ivHex,'hex');
  const tag = Buffer.from(tagHex,'hex');
  const encrypted = Buffer.from(encHex,'hex');
  const decipher = crypto.createDecipheriv(ALGO, KEY, iv);
  decipher.setAuthTag(tag);
  return Buffer.concat([decipher.update(encrypted), decipher.final()]).toString('utf8');
}

PART 2: AUTHENTICATION SYSTEM
═══════════════════════════════════════════════

SESSION-BASED AUTH (simpler and more secure than JWT for web apps):
Use express-session with SQLite store.

Session config:
app.use(session({
  secret: process.env.SESSION_SECRET,
  resave: false,
  saveUninitialized: false,
  store: new SQLiteStore({ db: 'sessions.db', dir: './data' }),
  cookie: {
    secure: process.env.NODE_ENV === 'production',
    httpOnly: true,       // XSS protection
    maxAge: 8 * 60 * 60 * 1000,  // 8 hours
    sameSite: 'strict'    // CSRF protection
  }
}));

AUTH ENDPOINTS:

POST /auth/login
Body: { email, password }
- Find user by email
- bcrypt.compare(password, user.password)
- If valid: req.session.userId = user.id, req.session.role = user.role
- Log to audit_log: action='login', ip=req.ip
- Return: { success: true, role: user.role, redirect: role==='admin' ? '/admin' : '/client' }
- On fail: wait 500ms (prevent timing attacks), return 401

POST /auth/logout  
- req.session.destroy()
- Clear cookie
- Redirect to /login

GET /auth/me
- Returns current session user (id, name, email, role, client_id)
- Used by frontend to determine which UI to show

MIDDLEWARE — protect all routes:

function requireAuth(req, res, next) {
  if (!req.session.userId) return res.status(401).json({ error: 'Unauthorized' });
  next();
}

function requireAdmin(req, res, next) {
  if (!req.session.userId || req.session.role !== 'admin') {
    return res.status(403).json({ error: 'Forbidden' });
  }
  next();
}

function requireClientAccess(req, res, next) {
  // Client can only access their own project
  // Admin can access any project
  const { projectId } = req.params;
  if (req.session.role === 'admin') return next();
  if (req.session.role === 'client') {
    const user = db.prepare('SELECT client_id FROM users WHERE id = ?')
                   .get(req.session.userId);
    if (user.client_id === projectId) return next();
    return res.status(403).json({ error: 'Access denied to this store' });
  }
  res.status(403).json({ error: 'Forbidden' });
}

Apply middleware:
- All /api/admin/* routes → requireAdmin
- All /api/client/* routes → requireAuth + requireClientAccess
- All /api/projects/:projectId/* → requireAuth + requireClientAccess

PART 4: CLIENT INVITATION FLOW
═══════════════════════════════════════════════

When admin creates a new project and wants to give client access:

STEP 1 — Admin clicks "Invitar cliente" on a project:
POST /api/admin/projects/:projectId/invite
Body: { email, name }
- Create user record: { role: 'client', client_id: projectId, is_active: 0 }
- Generate invite token: crypto.randomBytes(32).toString('hex')
- Store: invite_token (hashed), invite_expires (48h from now)
- Send email via Resend/Nodemailer:

Subject: "Acceso a tu panel de tienda — [StoreName]"

Body:
  Hola [ClientName],
  
  [AgencyName] te ha dado acceso a tu panel de gestión de tienda.
  Podrás ver las métricas, el rendimiento y el progreso de [StoreName]
  en tiempo real.
  
  Haz clic aquí para configurar tu acceso:
  [INVITE_LINK]
  
  Este enlace caduca en 48 horas.

STEP 2 — Client clicks link → GET /invite/:token
- Validate token (not expired, not used)
- Show password setup form (name pre-filled, set password)
- POST /invite/:token/setup { password }
- Hash password, activate user, clear invite token
- Auto-login, redirect to /client

- Cannot navigate to other stores
- Cannot access admin routes

PART 6: ROW-LEVEL SECURITY ENFORCEMENT
═══════════════════════════════════════════════

CRITICAL — implement these checks on EVERY data query:

// Helper: get project only if user has access
function getProjectSecure(projectId, session) {
  if (session.role === 'admin') {
    return db.prepare('SELECT * FROM projects WHERE id = ?').get(projectId);
  }
  if (session.role === 'client') {
    const user = db.prepare('SELECT client_id FROM users WHERE id = ?')
                   .get(session.userId);
    if (user.client_id !== projectId) throw new Error('Access denied');
    return db.prepare('SELECT * FROM projects WHERE id = ?').get(projectId);
  }
  throw new Error('Unauthorized');
}

// Never use: db.prepare('SELECT * FROM projects').all()  ← for clients
// Always use: getProjectSecure(projectId, req.session)   ← enforces isolation

AUDIT EVERY ACCESS:
Log every data access to audit_log table:
- Who accessed what
- When
- From which IP
- What action

This creates a full audit trail for GDPR compliance and security review.

DATA ISOLATION CHECKLIST (verify each):
