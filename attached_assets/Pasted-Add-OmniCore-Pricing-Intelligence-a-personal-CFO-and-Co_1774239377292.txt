CREATE TABLE agency_cost_structure (
  id TEXT PRIMARY KEY,
  
  -- YOUR FIXED MONTHLY COSTS
  cost_claude_api REAL DEFAULT 0,       -- Anthropic API monthly
  cost_replicate REAL DEFAULT 0,         -- Replicate monthly avg
  cost_hosting REAL DEFAULT 0,           -- Replit/server
  cost_stripe_fees REAL DEFAULT 0,       -- 1.4% + €0.25 per transaction
  cost_domain_tools REAL DEFAULT 0,      -- domains, tools, subscriptions
  cost_your_time_hourly REAL DEFAULT 0,  -- what YOUR hour is worth
  avg_hours_per_client REAL DEFAULT 0,   -- hours/month per client
  
  -- PER-SERVICE COSTS
  cost_per_image_generated REAL DEFAULT 0.25,  -- Replicate avg
  cost_per_audit_run REAL DEFAULT 0.08,         -- Claude tokens
  cost_per_boost_masivo REAL DEFAULT 2.50,      -- full run
  cost_per_seo_update REAL DEFAULT 0.15,        -- Claude tokens
  cost_per_ab_test_setup REAL DEFAULT 0.05,
  cost_per_report_pdf REAL DEFAULT 0.20,
  
  -- YOUR DESIRED MARGINS
  target_margin_setup REAL DEFAULT 0.70,      -- 70% margin on setup
  target_margin_retainer REAL DEFAULT 0.65,   -- 65% on monthly
  target_margin_extras REAL DEFAULT 0.75,     -- 75% on ad-hoc
  minimum_hourly_rate REAL DEFAULT 80,        -- min €/hr you accept
  
  updated_at TEXT DEFAULT (datetime('now'))
);

CREATE TABLE service_catalog (
  id TEXT PRIMARY KEY,
  service_name TEXT NOT NULL,
  service_type TEXT,  -- setup | retainer | extra | consultation
  
  -- COSTS (what it costs YOU)
  cost_time_hours REAL DEFAULT 0,
  cost_platform REAL DEFAULT 0,      -- API costs
  cost_tools REAL DEFAULT 0,
  total_cost REAL DEFAULT 0,         -- calculated
  
  -- PRICING (what you charge)
  price_current REAL,
  price_suggested REAL,              -- OmniCore suggestion
  price_min REAL,                    -- never go below this
  price_max REAL,                    -- market ceiling
  
  -- MARKET CONTEXT
  market_avg_price REAL,             -- what competitors charge
  market_price_low REAL,
  market_price_high REAL,
  our_positioning TEXT,              -- budget|mid|premium|luxury
  
  -- PERFORMANCE
  times_sold INTEGER DEFAULT 0,
  times_rejected INTEGER DEFAULT 0,  -- lost deals (track manually)
  avg_client_satisfaction REAL,
  
  -- OMNICORE ANALYSIS
  last_price_review TEXT,
  omnicore_recommendation TEXT,      -- latest AI recommendation
  omnicore_confidence REAL,
  price_change_suggested REAL,       -- delta from current
  
  is_active INTEGER DEFAULT 1,
  created_at TEXT DEFAULT (datetime('now'))
);

CREATE TABLE pricing_decisions (
  id TEXT PRIMARY KEY,
  decision_type TEXT,  -- product|service|saas_plan
  entity_id TEXT,      -- productId or service id or plan name
  entity_name TEXT,
  
  -- THE DECISION
  old_price REAL,
  new_price REAL,
  price_delta REAL,
  price_delta_pct REAL,
  
  -- OMNICORE REASONING (full transparency)
  reasoning TEXT,           -- complete justification
  data_used TEXT,           -- JSON: what data informed this
  confidence REAL,
  risk_level TEXT,          -- low|medium|high
  expected_impact TEXT,     -- what OmniCore predicts will happen
  
  -- OUTCOME TRACKING
  decision_made_at TEXT,
  decision_by TEXT,         -- omnicore_auto|admin_accepted|admin_override
  outcome_measured_at TEXT,
  actual_revenue_change REAL,
  actual_conversion_change REAL,
  prediction_accuracy REAL, -- how close was OmniCore's prediction
  
  created_at TEXT DEFAULT (datetime('now'))
);

AUTOMATED PRICING RULES ENGINE:
Admin sets rules once, OmniCore executes:

CREATE TABLE pricing_rules (
  id TEXT PRIMARY KEY,
  project_id TEXT NOT NULL,
  rule_name TEXT,
  rule_type TEXT,
  -- Types:
  -- margin_floor: never below X% margin
  -- competitor_match: match competitor ± X%
  -- velocity_based: price up when selling fast
  -- inventory_based: price down when overstocked
  -- seasonal: price X% higher in peak months
  -- psychological: always round to .99/.95/.90
  -- premium_positioning: always above competitor avg
  
  condition_json TEXT,   -- when to trigger
  action_json TEXT,      -- what price change to make
  limits_json TEXT,      -- min/max price limits
  
  is_active INTEGER DEFAULT 1,
  auto_apply INTEGER DEFAULT 0,  -- apply without asking
  requires_approval INTEGER DEFAULT 1,
  
  times_triggered INTEGER DEFAULT 0,
  last_triggered TEXT
);

EXAMPLE RULES:
Rule 1 — Margin Floor:
{
  name: "Nunca por debajo del 30% de margen",
  type: "margin_floor",
  condition: "IF calculated_margin < 0.30",
  action: "SET price = cogs_total / 0.70",
  auto_apply: true,
  requires_approval: false
}

Rule 2 — Velocity Pricing (trader mindset):
{
  name: "Sube precio si vende rápido",
  type: "velocity_based",
  condition: "IF daily_sales > avg_daily_sales * 1.5 FOR 3 days",
  action: "INCREASE price BY 8%",
  limits: { max_increase_pct: 25 },
  auto_apply: false,
  requires_approval: true,
  message: "Este producto está vendiendo 1.5x su media. 
            El mercado aguanta una subida de precio."
}

Rule 3 — Inventory Pressure:
{
  name: "Baja precio si hay exceso de stock",
  type: "inventory_based", 
  condition: "IF days_remaining > 90 AND sales_velocity = 'slow'",
  action: "DECREASE price BY 15%",
  auto_apply: false,
  message: "90+ días de stock. Una bajada del 15% 
            aceleraría rotación y libera capital."
}

Rule 4 — Competitor Monitoring:
{
  name: "Mantén precio 5% bajo competencia directa",
  type: "competitor_match",
  condition: "IF competitor_price_changed BY more than 5%",
  action: "SET price = competitor_median * 0.95",
  limits: { min_margin: 0.25 },
  requires_approval: true
}

Rule 5 — Psychological Always-On:
{
  name: "Siempre terminar en .99/.95/.90",
  type: "psychological",
  condition: "ALWAYS — on any price change",
  action: "ROUND to nearest psychological price point",
  auto_apply: true
}
