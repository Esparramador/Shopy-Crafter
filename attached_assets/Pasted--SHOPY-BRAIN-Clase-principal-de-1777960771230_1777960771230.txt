import Anthropic from '@anthropic-ai/sdk';
import { GoogleGenerativeAI } from '@google/generative-ai';
import { createHash } from 'crypto';
import {
  ALL_TEMPLATES,
  buildPromptFromTemplate,
  type ReportType,
  type PromptTemplate,
} from './templates';

// ============================================================
// TIPOS
// ============================================================

interface GenerateInputs {
  context: Record<string, unknown>;
  data: Record<string, unknown>;
  shopId: string | number;
  userId?: string | number;
}

interface GenerateOptions {
  forceFresh?: boolean;        // ignorar caché
  maxRetries?: number;         // por defecto 1
  fallbackToGemini?: boolean;  // por defecto true
  customModel?: string;        // override del modelo
}

interface GenerateResult<T> {
  data: T;
  metadata: {
    model: string;
    promptVersion: string;
    promptTokens: number;
    completionTokens: number;
    costEur: number;
    latencyMs: number;
    fromCache: boolean;
    confidence: number;
    warnings: string[];
  };
}

interface ValidationResult {
  valid: boolean;
  failures: string[];
}

// ============================================================
// CACHE INTERFACE — implementa según tu BD
// ============================================================

interface CacheStore {
  get(key: string): Promise<unknown | null>;
  set(key: string, value: unknown, ttlSeconds: number): Promise<void>;
}

// Implementación in-memory básica — en producción usa Redis o Postgres
class InMemoryCache implements CacheStore {
  private store = new Map<string, { value: unknown; expiresAt: number }>();

  async get(key: string): Promise<unknown | null> {
    const entry = this.store.get(key);
    if (!entry) return null;
    if (entry.expiresAt < Date.now()) {
      this.store.delete(key);
      return null;
    }
    return entry.value;
  }

  async set(key: string, value: unknown, ttlSeconds: number): Promise<void> {
    this.store.set(key, { value, expiresAt: Date.now() + ttlSeconds * 1000 });
  }
}

// ============================================================
// LOGGER INTERFACE — implementa contra tu BD
// ============================================================

interface AIGenerationLog {
  id: string;
  shopId: string | number;
  userId?: string | number;
  reportType: ReportType;
  modelUsed: string;
  promptVersion: string;
  promptTokens: number;
  completionTokens: number;
  costEur: number;
  latencyMs: number;
  success: boolean;
  errorMessage?: string;
  promptHash: string;
  outputSummary: string;
  createdAt: Date;
}

interface Logger {
  log(entry: AIGenerationLog): Promise<void>;
}

class ConsoleLogger implements Logger {
  async log(entry: AIGenerationLog): Promise<void> {
    console.log('[ShopyBrain]', JSON.stringify(entry, null, 2));
    // En producción: insertar en tabla ai_generations
  }
}

// ============================================================
// SHOPY BRAIN — clase principal
// ============================================================

export class ShopyBrain {
  private claude: Anthropic;
  private gemini: GoogleGenerativeAI;
  private cache: CacheStore;
  private logger: Logger;
  private readonly version = '1.0';

  constructor(options?: {
    cache?: CacheStore;
    logger?: Logger;
  }) {
    if (!process.env.ANTHROPIC_API_KEY) {
      throw new Error('ANTHROPIC_API_KEY is required');
    }

    this.claude = new Anthropic({ apiKey: process.env.ANTHROPIC_API_KEY });

    if (process.env.GEMINI_API_KEY) {
      this.gemini = new GoogleGenerativeAI(process.env.GEMINI_API_KEY);
    } else {
      console.warn('[ShopyBrain] GEMINI_API_KEY not set, fallback disabled');
    }

    this.cache = options?.cache ?? new InMemoryCache();
    this.logger = options?.logger ?? new ConsoleLogger();
  }

  // ============================================================
  // MÉTODO PRINCIPAL
  // ============================================================

  async generate<T = unknown>(
    reportType: ReportType,
    inputs: GenerateInputs,
    options: GenerateOptions = {}
  ): Promise<GenerateResult<T>> {
    const startTime = Date.now();
    const template = ALL_TEMPLATES[reportType];

    if (!template) {
      throw new Error(`Unknown report type: ${reportType}`);
    }

    // 1. Construir prompt
    const prompt = buildPromptFromTemplate(template, inputs);

    // 2. Validar prompt
    const validation = this.validatePrompt(template, inputs);
    if (!validation.valid) {
      throw new Error(`Invalid prompt for ${reportType}: ${validation.failures.join(', ')}`);
    }

    // 3. Generar hash del prompt para caché
    const promptHash = this.hashPrompt(prompt, inputs.shopId);

    // 4. Comprobar caché
    if (!options.forceFresh) {
      const cached = await this.cache.get(promptHash);
      if (cached) {
        return {
          data: cached as T,
          metadata: {
            model: 'cache',
            promptVersion: template.version,
            promptTokens: 0,
            completionTokens: 0,
            costEur: 0,
            latencyMs: Date.now() - startTime,
            fromCache: true,
            confidence: 1.0,
            warnings: [],
          },
        };
      }
    }

    // 5. Llamar al modelo con retry y fallback
    const { result, modelUsed, tokens, cost } = await this.callWithFallback<T>(
      template,
      prompt,
      options
    );

    // 6. Validar output
    const outputValidation = this.validateOutput(result, template);
    if (!outputValidation.valid) {
      throw new Error(`Invalid output structure: ${outputValidation.failures.join(', ')}`);
    }

    // 7. Guardar en caché
    await this.cache.set(promptHash, result, template.cacheTTLHours * 3600);

    // 8. Loggear
    const latencyMs = Date.now() - startTime;
    await this.logger.log({
      id: this.generateId(),
      shopId: inputs.shopId,
      userId: inputs.userId,
      reportType,
      modelUsed,
      promptVersion: template.version,
      promptTokens: tokens.input,
      completionTokens: tokens.output,
      costEur: cost,
      latencyMs,
      success: true,
      promptHash,
      outputSummary: this.summarizeOutput(result),
      createdAt: new Date(),
    });

    // 9. Devolver con metadata
    const meta = (result as { metadata?: { confidence?: number; warnings?: string[] } }).metadata;
    return {
      data: result,
      metadata: {
        model: modelUsed,
        promptVersion: template.version,
        promptTokens: tokens.input,
        completionTokens: tokens.output,
        costEur: cost,
        latencyMs,
        fromCache: false,
        confidence: meta?.confidence ?? 0,
        warnings: meta?.warnings ?? [],
      },
    };
  }

  // ============================================================
  // VALIDACIÓN DE PROMPT
  // ============================================================

  private validatePrompt(template: PromptTemplate, inputs: GenerateInputs): ValidationResult {
    const failures: string[] = [];

    if (!template.role || template.role.length < 50) {
      failures.push('Role demasiado corto o ausente');
    }
    if (!inputs.context || Object.keys(inputs.context).length < 3) {
      failures.push('Contexto insuficiente (mínimo 3 campos)');
    }
    if (!inputs.data || typeof inputs.data !== 'object') {
      failures.push('Datos no estructurados');
    }
    if (!template.task || !template.task.match(/^(Genera|Analiza|Recomienda|Compara|Estima|Resume|Diagnostica)/)) {
      failures.push('Task no empieza con verbo válido');
    }
    if (!template.constraints || template.constraints.length < 3) {
      failures.push('Mínimo 3 restricciones requeridas');
    }
    if (!template.outputFormat) {
      failures.push('OutputFormat no definido');
    }

    return { valid: failures.length === 0, failures };
  }

  // ============================================================
  // VALIDACIÓN DE OUTPUT
  // ============================================================

  private validateOutput(output: unknown, template: PromptTemplate): ValidationResult {
    const failures: string[] = [];

    if (typeof output !== 'object' || output === null) {
      failures.push('Output no es objeto');
      return { valid: false, failures };
    }

    const out = output as Record<string, unknown>;

    if (!out.metadata || typeof out.metadata !== 'object') {
      failures.push('Falta metadata');
    }

    // Verificar campos top-level del outputFormat
    const expectedKeys = Object.keys(template.outputFormat);
    for (const key of expectedKeys) {
      if (!(key in out)) {
        failures.push(`Falta campo: ${key}`);
      }
    }

    return { valid: failures.length === 0, failures };
  }

  // ============================================================
  // LLAMADA CON FALLBACK
  // ============================================================

  private async callWithFallback<T>(
    template: PromptTemplate,
    prompt: string,
    options: GenerateOptions
  ): Promise<{
    result: T;
    modelUsed: string;
    tokens: { input: number; output: number };
    cost: number;
  }> {
    const maxRetries = options.maxRetries ?? 1;

    // Intento 1: Claude
    for (let attempt = 0; attempt <= maxRetries; attempt++) {
      try {
        return await this.callClaude<T>(prompt, options.customModel);
      } catch (err) {
        console.warn(`[ShopyBrain] Claude attempt ${attempt + 1} failed:`, err);
        if (attempt < maxRetries) {
          await this.sleep(2000 * (attempt + 1)); // backoff exponencial
        }
      }
    }

    // Fallback: Gemini
    if (options.fallbackToGemini !== false && this.gemini) {
      console.warn('[ShopyBrain] Falling back to Gemini');
      try {
        return await this.callGemini<T>(prompt);
      } catch (err) {
        console.error('[ShopyBrain] Gemini also failed:', err);
        throw new Error('Both Claude and Gemini failed to generate output');
      }
    }

    throw new Error('Claude failed and Gemini fallback not available');
  }

  // ============================================================
  // CLAUDE
  // ============================================================

  private async callClaude<T>(
    prompt: string,
    customModel?: string
  ): Promise<{
    result: T;
    modelUsed: string;
    tokens: { input: number; output: number };
    cost: number;
  }> {
    const model = customModel || 'claude-opus-4-7';

    const response = await this.claude.messages.create({
      model,
      max_tokens: 4000,
      messages: [{ role: 'user', content: prompt }],
    });

    const textBlock = response.content.find((b) => b.type === 'text');
    if (!textBlock || textBlock.type !== 'text') {
      throw new Error('Claude returned no text content');
    }

    const text = textBlock.text.trim();
    const cleaned = this.stripMarkdownFences(text);

    let result: T;
    try {
      result = JSON.parse(cleaned) as T;
    } catch (err) {
      throw new Error(`Claude returned invalid JSON: ${(err as Error).message}\n\nRaw: ${cleaned.slice(0, 500)}`);
    }

    // Coste aproximado Claude Opus 4.7: $15/1M input, $75/1M output (consultar precios actuales)
    const costUsd =
      (response.usage.input_tokens * 15) / 1_000_000 +
      (response.usage.output_tokens * 75) / 1_000_000;
    const costEur = costUsd * 0.92;

    return {
      result,
      modelUsed: model,
      tokens: {
        input: response.usage.input_tokens,
        output: response.usage.output_tokens,
      },
      cost: costEur,
    };
  }

  // ============================================================
  // GEMINI (FALLBACK)
  // ============================================================

  private async callGemini<T>(prompt: string): Promise<{
    result: T;
    modelUsed: string;
    tokens: { input: number; output: number };
    cost: number;
  }> {
    if (!this.gemini) {
      throw new Error('Gemini not configured');
    }

    const model = this.gemini.getGenerativeModel({
      model: 'gemini-2.5-pro',
      generationConfig: {
        responseMimeType: 'application/json',
        maxOutputTokens: 4000,
      },
    });

    const response = await model.generateContent(prompt);
    const text = response.response.text().trim();
    const cleaned = this.stripMarkdownFences(text);

    let result: T;
    try {
      result = JSON.parse(cleaned) as T;
    } catch (err) {
      throw new Error(`Gemini returned invalid JSON: ${(err as Error).message}`);
    }

    const usage = response.response.usageMetadata;
    const inputTokens = usage?.promptTokenCount ?? 0;
    const outputTokens = usage?.candidatesTokenCount ?? 0;

    // Coste aproximado Gemini 2.5 Pro: $1.25/1M input, $5/1M output
    const costUsd = (inputTokens * 1.25) / 1_000_000 + (outputTokens * 5) / 1_000_000;
    const costEur = costUsd * 0.92;

    return {
      result,
      modelUsed: 'gemini-2.5-pro',
      tokens: { input: inputTokens, output: outputTokens },
      cost: costEur,
    };
  }

  // ============================================================
  // UTILIDADES
  // ============================================================

  private stripMarkdownFences(text: string): string {
    return text
      .replace(/^```(?:json)?\s*/i, '')
      .replace(/\s*```\s*$/, '')
      .trim();
  }

  private hashPrompt(prompt: string, shopId: string | number): string {
    return createHash('sha256').update(`${shopId}:${prompt}`).digest('hex').slice(0, 32);
  }

  private generateId(): string {
    return `gen_${Date.now()}_${Math.random().toString(36).slice(2, 10)}`;
  }

  private summarizeOutput(output: unknown): string {
    try {
      const str = JSON.stringify(output);
      return str.slice(0, 200);
    } catch {
      return 'unsummarizable';
    }
  }

  private sleep(ms: number): Promise<void> {
    return new Promise((resolve) => setTimeout(resolve, ms));
  }
}

// SINGLETON EXPORT
// ============================================================

export const shopyBrain = new ShopyBrain();
