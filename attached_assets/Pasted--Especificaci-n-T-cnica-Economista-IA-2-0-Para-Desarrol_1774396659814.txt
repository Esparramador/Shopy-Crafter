FILTRAR DATOS VÁLIDOS const validData = priceHistory.filter(entry => entry.sessions >= 20 && // mínimo de tráfico entry.price > 0 ); if (validData.length < minDataPoints) { return { coefficient: 0, confidence: 0, interpretation: "Datos insuficientes (mínimo 10 períodos con cambios de precio)", recommendation: "Espera más datos o realiza un test A/B de precio" }; } // 2. CALCULAR CAMBIOS PORCENTUALES const changes = []; for (let i = 1; i < validData.length; i++) { const prev = validData[i - 1]; const curr = validData[i]; const priceChange = (curr.price - prev.price) / prev.price; const demandChange = (curr.units_sold - prev.units_sold) / (prev.units_sold || 1); if (Math.abs(priceChange) > 0.02) { // solo cambios >2% changes.push({ priceChange, demandChange, elasticity: demandChange / priceChange }); } } // 3. CALCULAR ELASTICIDAD PROMEDIO const avgElasticity = changes.reduce((sum, c) => sum + c.elasticity, 0) / changes.length; // 4. CALCULAR CONFIANZA (basado en consistencia) const variance = changes.reduce((sum, c) => sum + Math.pow(c.elasticity - avgElasticity, 2), 0 ) / changes.length; const stdDev = Math.sqrt(variance); const confidence = Math.max(0, 100 - (stdDev * 50)); // menos varianza = más confianza // 5. INTERPRETACIÓN let interpretation = ""; let recommendation = ""; if (avgElasticity < -1.5) { interpretation = "MUY ELÁSTICO: Los clientes son muy sensibles al precio"; recommendation = "⚠️ Evita subir precio. Considera bajar 5-10% para aumentar volumen"; } else if (avgElasticity < -1.0) { interpretation = "ELÁSTICO: La demanda reacciona significativamente al precio"; recommendation = "💡 Prueba descuentos temporales para aumentar ventas"; } else if (avgElasticity < -0.5) { interpretation = "MODERADO: Demanda equilibrada respecto al precio"; recommendation = "✅ Puedes ajustar precio ±10% sin gran impacto"; } else { interpretation = "INELÁSTICO: Los clientes compran independiente del precio"; recommendation = "🚀 OPORTUNIDAD: Sube precio 15-25% para maximizar margen"; } return { coefficient: avgElasticity, confidence: Math.round(confidence), interpretation, recommendation }; } --- ## 🎯 **FASE 3: PRECIO ÓPTIMO CON IA** ### **Motor de recomendación de precio:** ``typescript
