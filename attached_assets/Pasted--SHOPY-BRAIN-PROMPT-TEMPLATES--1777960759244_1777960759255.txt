// ============================================================
// HELPER: Construir prompt completo desde template + inputs
// ============================================================

export function buildPromptFromTemplate(
  template: PromptTemplate,
  inputs: { context: Record<string, unknown>; data: Record<string, unknown> }
): string {
  const blocks = [
    template.role,
    '',
    'CONTEXTO:',
    Object.entries(inputs.context)
      .map(([k, v]) => `${k.toUpperCase()}: ${typeof v === 'object' ? JSON.stringify(v) : v}`)
      .join('\n'),
    '',
    'DATOS:',
    JSON.stringify(inputs.data, null, 2),
    '',
    'TAREA:',
    template.task,
    '',
    'RESTRICCIONES INVIOLABLES:',
    template.constraints.map((c, i) => `${i + 1}. ${c}`).join('\n'),
    '',
    'OUTPUT_FORMAT:',
    'Devuelve ÚNICAMENTE un JSON válido siguiendo esta estructura exacta.',
    'NO incluyas texto antes ni después. NO uses bloques markdown.',
    '',
    JSON.stringify(template.outputFormat, null, 2),
  ];

  if (template.examples && template.examples.length > 0) {
    blocks.push('', 'EJEMPLOS:', JSON.stringify(template.examples, null, 2));
  }

  return blocks.join('\n');
}
